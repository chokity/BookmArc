#!/usr/bin/env bash
# Arc for Firefox: give Arc the whole window (macOS & Linux).
#
# Adds Arc's layout to the Firefox profile that has Arc installed:
#   <profile>/chrome/userChrome.css  hides Firefox's tab strip and toolbar
#   <profile>/user.js                turns the stylesheet on, horizontal tabs,
#                                    opens Arc at startup
#   <profile>/customKeys.json        clears Firefox's Cmd/Ctrl+T and Cmd/Ctrl+W
#                                    so Arc can handle them
# Your own CSS and prefs are kept: Arc only manages its marked block.
#
# Usage:  bash arc-setup.sh              set up (or update)
#         bash arc-setup.sh --uninstall  remove Arc's blocks again
# Generated by scripts/build-setup.mjs; edit extras/ instead.
set -euo pipefail

ADDON_ID="arc@sidebar-pliha"
CSS_BEGIN="/* >>> Arc for Firefox >>> */"
CSS_END="/* <<< Arc for Firefox <<< */"
JS_BEGIN="// >>> Arc for Firefox >>>"
JS_END="// <<< Arc for Firefox <<<"

UNINSTALL=0
[ "${1:-}" = "--uninstall" ] && UNINSTALL=1

case "$(uname -s)" in
  Darwin) ROOTS=("$HOME/Library/Application Support/Firefox/Profiles") ;;
  *) ROOTS=("$HOME/.mozilla/firefox" "$HOME/snap/firefox/common/.mozilla/firefox" "$HOME/.var/app/org.mozilla.firefox/.mozilla/firefox") ;;
esac

# Every Firefox profile folder (has prefs.js); Arc's first.
ARC=()
OTHER=()
for root in "${ROOTS[@]}"; do
  [ -d "$root" ] || continue
  for dir in "$root"/*/; do
    dir="${dir%/}"
    [ -f "$dir/prefs.js" ] || continue
    if grep -q "\"id\":\"$ADDON_ID\"" "$dir/extensions.json" 2>/dev/null; then ARC+=("$dir"); else OTHER+=("$dir"); fi
  done
done

if [ ${#ARC[@]} -gt 0 ]; then
  CANDIDATES=("${ARC[@]}")
else
  echo "No Firefox profile with Arc installed was found. Install Arc first, or pick a profile:"
  CANDIDATES=("${OTHER[@]}")
fi
[ ${#CANDIDATES[@]} -gt 0 ] || { echo "No Firefox profiles found." >&2; exit 1; }

PROFILE="${CANDIDATES[0]}"
if [ ${#CANDIDATES[@]} -gt 1 ]; then
  echo "Several Firefox profiles found:"
  i=1
  for p in "${CANDIDATES[@]}"; do echo "  $i) $(basename "$p")"; i=$((i + 1)); done
  printf "Which one? [1-%s] " "${#CANDIDATES[@]}"
  read -r pick < /dev/tty
  case "$pick" in ''|*[!0-9]*) echo "Not a number." >&2; exit 1 ;; esac
  [ "$pick" -ge 1 ] && [ "$pick" -le ${#CANDIDATES[@]} ] || { echo "Out of range." >&2; exit 1; }
  PROFILE="${CANDIDATES[$((pick - 1))]}"
fi

# Replace (or remove) the marked block in a file, keeping everything else.
write_block() {
  local file="$1" begin="$2" end="$3" content="$4" tmp
  mkdir -p "$(dirname "$file")"
  # A symlinked file belongs to something else (a dotfiles repo, or Arc's own
  # source in a development profile): writing the block would edit that.
  if [ -L "$file" ]; then
    echo "Skipping $file: it is a symlink, so it is managed elsewhere."
    return 0
  fi
  touch "$file"
  tmp="$(mktemp)"
  # Drop the old block (and trailing blank lines), keep everything else.
  awk -v b="$begin" -v e="$end" '
    index($0, b) { skip = 1 }
    !skip { if ($0 == "") blank++; else { while (blank > 0) { print ""; blank-- } print } }
    index($0, e) { skip = 0 }' "$file" > "$tmp"
  if [ "$UNINSTALL" -eq 0 ]; then
    [ -s "$tmp" ] && printf '\n' >> "$tmp"
    printf '%s\n%s\n%s\n' "$begin" "$content" "$end" >> "$tmp"
  fi
  cat "$tmp" > "$file"
  rm -f "$tmp"
}

# read -d '' (not $(cat <<...)): macOS bash 3.2 misparses quotes inside $( heredoc ).
IFS= read -r -d '' CSS <<'ARC_EOF' || true
/*
 * ArcFox layout for Firefox. Always on: ArcFox's sidebar replaces the tab strip.
 * Firefox's toolbar (back/forward, address bar, extension buttons) stays on top,
 * and the window keeps its native title bar (browser.tabs.inTitlebar = 0 in
 * extras/user.js) for the window buttons.
 *
 * Setup (pair with extras/user.js, which also opens ArcFox at startup):
 *   1. about:support -> Profile Folder -> Open. Create a "chrome" folder there.
 *   2. Copy this file into it as chrome/userChrome.css.
 *   3. Copy extras/user.js into the profile folder, then restart Firefox.
 */

:root {
  /* Starting width only: drag the edge to resize, Firefox remembers it. */
  --arcfox-sidebar-width: 260px;
  --arcfox-sidebar-min: 180px;
  --arcfox-sidebar-max: 620px;
}

/* ---------- Arc's background, for Firefox's own panels ---------- */

/* Arc tints its sidebar with the current space's color. Firefox's panels are
   chrome and can't read that, so Arc writes the color into the window title
   as an invisible marker (background.js) and it's decoded here. Values mirror
   shared/theme.css. */
:root {
  --arcfox-base: #efedf5;
  --arcfox-tint: 16%;
  --arcfox-space: #af51f5;
  --arcfox-panel-bg: color-mix(in srgb, var(--arcfox-space) var(--arcfox-tint), var(--arcfox-base));
}

@media (prefers-color-scheme: dark) {
  :root {
    --arcfox-base: #1c1a22;
    --arcfox-tint: 12%;
  }
}

#main-window[titlepreface*="\200C"] { --arcfox-space: #37adff; } /* blue */
#main-window[titlepreface*="\200D"] { --arcfox-space: #00c79a; } /* turquoise */
#main-window[titlepreface*="\2060"] { --arcfox-space: #51cd00; } /* green */
#main-window[titlepreface*="\2061"] { --arcfox-space: #ffcb00; } /* yellow */
#main-window[titlepreface*="\2062"] { --arcfox-space: #ff9f00; } /* orange */
#main-window[titlepreface*="\2063"] { --arcfox-space: #ff613d; } /* red */
#main-window[titlepreface*="\2064"] { --arcfox-space: #ff4bda; } /* pink */
#main-window[titlepreface*="\FEFF"] { --arcfox-space: #af51f5; } /* purple */

/* The panel area carries the color; Firefox's panels then let it through. */
#sidebar-box,
#sidebar {
  background-color: var(--arcfox-panel-bg) !important;
}
:root {
  --sidebar-background-color: var(--arcfox-panel-bg) !important;
}

@-moz-document url-prefix("chrome://browser/content/places/"),
  url-prefix("chrome://browser/content/syncedtabs/"),
  url-prefix("chrome://browser/content/genai/"),
  url-prefix("chrome://browser/content/sidebar/") {
  :root,
  body,
  #sidebar-panel-header,
  #history-panel,
  #bookmarksPanel,
  #syncedTabsContent,
  tree,
  tree > treechildren {
    background: transparent !important;
    background-color: transparent !important;
  }
}

/* ---------- Firefox chrome ArcFox replaces ---------- */

/* Firefox's sidebar launcher and the panel header ("Arc  x"):
   #sidebar-header in the old sidebar; #sidebar-panel-header in the new one,
   which lives inside the panel document (webext-panels.xhtml). */
#sidebar-container,
#sidebar-launcher-splitter,
#sidebar-header {
  visibility: collapse !important;
  display: none !important;
}

/* Arc's sidebar is resizable: Firefox's splitter stays, as an invisible grab
   strip on the edge (it draws the border Arc removed otherwise). Firefox
   remembers the width it's dragged to. */
#sidebar-splitter {
  /* No display override: a <splitter> drags because of its own frame, and
     forcing a CSS display (flex) replaces that frame — the strip then looks
     right but stops resizing once the page lays out. Visibility is stated
     because an older copy of this file collapses the splitter. */
  visibility: visible !important;
  appearance: none !important;
  width: 8px !important;
  min-width: 8px !important;
  /* Entirely over the sidebar: the half that used to hang into the content
     area sits above a remote page, which swallows the drag once it loads. */
  margin-inline-start: -8px !important;
  margin-inline-end: 0 !important;
  pointer-events: auto !important;
  border: 0 !important;
  background: transparent !important;
  cursor: col-resize !important;
  z-index: 11 !important;
}

/* Panel closed: no edge to drag, and the strip would otherwise sit over the
   page and swallow clicks. */
#sidebar-box[hidden] + #sidebar-splitter {
  display: none !important;
}

/* The panel header belongs to the panel's own document, and userChrome.css
   styles every chrome document — so hide it only in Arc's panel. Firefox's
   own panels (bookmarks, history, synced tabs, AI chat) keep their title
   and close button. */
@-moz-document url-prefix("chrome://browser/content/webext-panels.xhtml") {
  #sidebar-panel-header {
    visibility: collapse !important;
    display: none !important;
  }
}

/* Newer Firefox puts the panel header in the browser window, right above the
   panel: hide it while the sidebar shows Arc's panel. */
#sidebar-box:has(> .sidebar-browser-stack > #sidebar[src*="webext-panels"]) > #sidebar-panel-header {
  visibility: collapse !important;
  display: none !important;
}

/* The tab strip: Arc lists the tabs. The window buttons live in the native
   title bar (browser.tabs.inTitlebar = 0), so the whole strip can go. */
#TabsToolbar {
  visibility: collapse !important;
}

/* Sharp edge between Arc and the page: Firefox rounds and shadows both the
   sidebar panel and the content area. */
#sidebar,
#sidebar-box,
.sidebar-browser-stack,
#tabbrowser-tabbox,
#tabbrowser-tabpanels,
.browserSidebarContainer,
.browserContainer,
.browserStack,
#tabbrowser-tabpanels browser {
  border-radius: 0 !important;
  border: none !important;
  outline: none !important;
  box-shadow: none !important;
  margin: 0 !important;
}

/* Arc's page card: while the sidebar is open, the page gets a rounded top-left
   corner and a soft shadow falling onto the sidebar. The sidebar reaches 8px
   under the page (its own padding), so the rounded corner shows the sidebar's
   real color, custom theme colors included. */
#main-window:has(#sidebar-box:not([hidden])) #sidebar-box {
  margin-inline-end: -8px !important;
}
#main-window:has(#sidebar-box:not([hidden])) #tabbrowser-tabbox {
  position: relative !important;
  z-index: 1 !important;
  border-radius: 8px 0 0 0 !important;
  overflow: clip !important;
  box-shadow: 0 0 0 0.5px rgba(0, 0, 0, 0.14), 0 1px 8px rgba(0, 0, 0, 0.16) !important;
}

/* Firefox leaves a gap under the sidebar (padding-block-end); Arc fills it. */
#sidebar-box,
#sidebar-container,
.sidebar-browser-stack {
  padding: 0 !important;
}

/* The separator line itself (Firefox draws it as a border on the sidebar). */
:root {
  --sidebar-border-color: transparent !important;
  --chrome-content-separator-color: transparent !important;
}

/* Starting width and limits for Arc's panel. */
#sidebar-box {
  width: var(--arcfox-sidebar-width);
  min-width: var(--arcfox-sidebar-min) !important;
  max-width: var(--arcfox-sidebar-max) !important;
}

/* ---------- Firefox's own sidebar panels ---------- */

/* When the sidebar shows one of Firefox's panels instead of Arc's (which
   Firefox loads as webext-panels.xhtml): same width as Arc's own panel (drag
   the edge to change both). Firefox asks for 400px for the AI chat; Arc's
   width wins. */
#main-window:has(#sidebar[src]:not([src*="webext-panels"]):not([src="about:blank"])) #sidebar-box {
  min-width: var(--arcfox-sidebar-min) !important;
}

/* Arc's own small windows (command bar, space editor) are popups: no toolbar. */
#main-window[chromehidden~="toolbar"] #nav-bar {
  display: none !important;
}
ARC_EOF
CSS="${CSS%$'\n'}"

IFS= read -r -d '' PREFS <<'ARC_EOF' || true
// ArcFox: copy into your Firefox profile folder (next to prefs.js).
// Firefox applies these on every start.

// Load chrome/userChrome.css.
user_pref("toolkit.legacyUserProfileCustomizations.stylesheets", true);
// Horizontal tabs (their strip is hidden; ArcFox lists the tabs).
user_pref("sidebar.verticalTabs", false);
// Native title bar: it holds the window buttons once the tab strip is hidden.
user_pref("browser.tabs.inTitlebar", 0);
// Closing the last tab keeps the window (with a new tab), like Arc.
user_pref("browser.tabs.closeWindowWithLastTab", false);
// A video pops out into Picture-in-Picture when you leave its tab, and goes back
// when you return, like Arc. Firefox has this built in; some profiles have it off.
user_pref("media.videocontrols.picture-in-picture.enabled", true);
user_pref("media.videocontrols.picture-in-picture.enable-when-switching-tabs.enabled", true);
// Reopen windows and tabs, including the open ArcFox sidebar.
user_pref("browser.startup.page", 3);
// Arc's panel state (open, and how wide) lives in sidebar.backupState, which
// Firefox writes itself. The setup seeds it once in prefs.js instead of here:
// a user_pref would be re-applied at every start, throwing away a width the
// sidebar had been dragged to.
ARC_EOF
PREFS="${PREFS%$'\n'}"

# Firefox's own Cmd/Ctrl+T and Cmd/Ctrl+W are reserved shortcuts, so Arc can
# only have them once the built-in keys are cleared. An empty entry in
# customKeys.json means "cleared"; other entries there are left alone.
# Injected verbatim into the setup scripts by scripts/build-setup.mjs.
# (.txt because add-on review flags .sh files.)
KEYS="$PROFILE/customKeys.json"
ARC_KEYS="key_newNavigatorTab key_close"

clear_default_key() { # $1 = key element id
  if [ ! -s "$KEYS" ] || ! grep -q '[^[:space:]{}]' "$KEYS"; then
    printf '{"%s":{}}' "$1" > "$KEYS"
  elif ! grep -q "$1" "$KEYS"; then
    tmp="$(mktemp)"
    awk -v k="$1" 'NR == 1 { sub(/^[ \t]*\{/, "{\"" k "\":{},") } 1' "$KEYS" > "$tmp"
    cat "$tmp" > "$KEYS"
    rm -f "$tmp"
  fi
}

restore_default_key() { # $1 = key element id
  [ -f "$KEYS" ] || return 0
  tmp="$(mktemp)"
  awk -v k="$1" '{
    pat = "\"" k "\"[ \t]*:[ \t]*\\{\\}"
    gsub(pat "[ \t]*,[ \t]*", "")
    gsub("[ \t]*,[ \t]*" pat, "")
    gsub(pat, "")
    print
  }' "$KEYS" > "$tmp"
  cat "$tmp" > "$KEYS"
  rm -f "$tmp"
}

for key in $ARC_KEYS; do
  if [ "$UNINSTALL" -eq 0 ]; then clear_default_key "$key"; else restore_default_key "$key"; fi
done

# An Arc stylesheet outside Arc's block is an older copy from a manual
# install: it keeps applying, and rules the new block doesn't restate win.
if grep -q "arcfox-sidebar-width" "$PROFILE/chrome/userChrome.css" 2>/dev/null &&
   ! grep -q "$CSS_BEGIN" "$PROFILE/chrome/userChrome.css" 2>/dev/null; then
  echo "Note: chrome/userChrome.css already has an Arc stylesheet outside Arc's block."
  echo "      Remove that older copy, or the two will fight."
fi

write_block "$PROFILE/chrome/userChrome.css" "$CSS_BEGIN" "$CSS_END" "$CSS"
write_block "$PROFILE/user.js" "$JS_BEGIN" "$JS_END" "$PREFS"

# Open Arc's panel on a profile that has never had it open. This goes into
# prefs.js, which Firefox owns from then on: it records the width the sidebar
# is dragged to, and a user_pref would overwrite that at every start.
# Injected verbatim into the setup scripts by scripts/build-setup.mjs.
# (.txt because add-on review flags ".sh" in a file name.)
SEED='user_pref("sidebar.backupState", "{\"command\":\"arc_sidebar-pliha-sidebar-action\",\"panelOpen\":true,\"launcherVisible\":false,\"launcherExpanded\":false}");'
if [ "$UNINSTALL" -eq 0 ] && [ -f "$PROFILE/prefs.js" ] && ! grep -q 'sidebar.backupState' "$PROFILE/prefs.js"; then
  printf '%s\n' "$SEED" >> "$PROFILE/prefs.js"
  echo "Seeded sidebar.backupState so Arc's panel opens; Firefox keeps its width from now on."
fi

if [ "$UNINSTALL" -eq 1 ]; then
  echo "Removed Arc's layout from $(basename "$PROFILE")."
  echo "Restart Firefox. user.js prefs already applied stay until you change them in about:config."
else
  echo "Arc's layout is set up in $(basename "$PROFILE")."
  echo "Quit Firefox completely and open it again."
fi
